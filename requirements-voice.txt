# Voice pipeline dependencies - installed separately because they are heavier
# and involve native builds that take longer on a Raspberry Pi.
#
# openwakeword is intentionally NOT listed here as a normal requirement: its
# published metadata unconditionally depends on `tflite-runtime`, which has
# no prebuilt wheel for recent Python versions (e.g. Python 3.12+, as
# shipped by Debian trixie / current Raspberry Pi OS) on aarch64.
# alex/voice/wakeword.py only ever uses openwakeword's ONNX backend
# (inference_framework="onnx"), so tflite-runtime is never actually needed
# at runtime - see scripts/install_raspberry_pi.sh, which installs
# openwakeword with `--no-deps` after these packages (its real runtime
# dependencies) are already satisfied.
onnxruntime>=1.10.0,<2
scipy>=1.3,<2
scikit-learn>=1,<2
requests>=2.0,<3
tqdm>=4,<5

faster-whisper>=1.0
piper-tts>=1.2
